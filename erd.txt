Table generator_types {
  generator_type_id string [pk, default: `generate_uuid()`, not null]
  name string [unique, not null]
  description string [not null]
  cost_data int [default: 0, not null]
  cost_high int [default: 0, not null]
}

Table generators {
  generator_id string [pk, default: `generate_uuid()`, not null]
  generator_type_id string [not null]
  owner_id string [not null]
  level int [default: 1, not null]
  x_position int [not null]
  world_position int [not null]
  isdeveloping boolean [default: false, not null]
  build_complete_ts int
  heat int [default: 0, not null]
  running boolean [default: true, not null]
}

Table map_progress {
  map_progress_id string [pk, default: `generate_uuid()`, not null]

  production_upgrade int [default: 0, not null]
  heat_reduction_upgrade int [default: 0, not null]
  tolerance_upgrade int [default: 0, not null]

  Indexes {
    (user_id, generator_id) [unique]
  }
}

Table inquiries {
  inquiry_id string [pk, default: `generate_uuid()`, not null]

  created_at bigint [not null, note: 'timestamp in milliseconds']
}
